        let isKurdish = true;
        let phraseIndex = 0;
        let charIndex = 0;
        let isDeleting = false;
        let typingTimeout = null;

        function runTyping() {
            const currentList = isKurdish ? phrasesKu : phrasesEn;
            const currentPhrase = currentList[phraseIndex];
            const typingElem = document.getElementById("typing");

            if (isDeleting) {
                typingElem.textContent = currentPhrase.substring(0, charIndex - 1);
                charIndex--;
            } else {
                typingElem.textContent = currentPhrase.substring(0, charIndex + 1);
                charIndex++;
            }

            let typeSpeed = isDeleting ? 35 : 85;

            if (!isDeleting && charIndex === currentPhrase.length) {
                typeSpeed = 1900; // Pause at end of phrase
                isDeleting = true;
            } else if (isDeleting && charIndex === 0) {
                isDeleting = false;
                phraseIndex = (phraseIndex + 1) % currentList.length;
                typeSpeed = 350; // Pause before typing new phrase
            }

            typingTimeout = setTimeout(runTyping, typeSpeed);
        }

        function restartTyping() {
            clearTimeout(typingTimeout);
            charIndex = 0;
            isDeleting = false;
            document.getElementById("typing").textContent = "";
            runTyping();
        }
